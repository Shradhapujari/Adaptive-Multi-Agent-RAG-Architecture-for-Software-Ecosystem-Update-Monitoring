# Pinned so a re-run reproduces the reported numbers. The build scripts and the
# eval harness both claim byte-identical output on replay; unpinned ranges break

#
# Versions below are the ones present in venv311 and used for every number in
# results/ -- except streamlit, which is NOT installed in that venv (the demo
# UI is not exercised by the harness or the tests). Verify before quoting it.
requests==2.34.2

pytest==9.1.1

# zoneinfo (stdlib) needs the IANA tz database. macOS and most desktop Linux
# ship it at the OS level, so temporal.py's ZoneInfo("America/Los_Angeles")
# works here without this package -- but Streamlit Community Cloud's base
# image does not, and raised ZoneInfoNotFoundError on import, which crashed
# the whole app before a single line of it ran. tzdata is the officially
# recommended fallback: a pure-data package with no system dependency.
tzdata==2026.4

# Demo UI only (app_1.py, DemoMain*.py, marag_app.py -- NOT app.py, which is
# an unrelated Flask IoT sensor API). app_1.py is the Streamlit deploy target;
# exercised end to end locally under Python 3.11 + streamlit 1.62.0 against the
# live releasetrain.io endpoints.
#
# Both of its model-dependent steps degrade without one: the query rewrite falls
# back to rule-based expansion, and the Answer Presenter composes its cited
# paragraph by rule and says so in the UI. So a host with no reachable Ollama and
# no API key (e.g. Streamlit Community Cloud) runs without a crash or a hidden
# failure -- just a blunter rewrite and a blunter answer. Temporal grounding
# (temporal.py) is rule-based by design and needs no model at all.
streamlit>=1.32.0
